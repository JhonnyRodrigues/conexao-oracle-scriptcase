// Contrato comum dos parametros:
// SQL: :nome. Array: ['nome' => $valor], sem ':' na chave.
// Nomes: [A-Za-z][A-Za-z0-9_]*. Valores: string, int, float ou null.
// Nao aceita bool, arrays, objetos ou chaves numericas.
// Binds representam valores; nomes de tabela/coluna exigem opcoes permitidas.
// SQL e parametros seguem separados ao driver, sem substituicao textual.
// Datas/decimais exigem formatos e conversoes SQL explicitos (exemplo no fim).
// Tipos retornados para numeros/datas dependem do driver; nao ha conversao extra.

// Falhas:
// InvalidArgumentException: configuracao, conexao ou parametros invalidos.
// UnexpectedValueException: texto invalido ou conversao com perda.
// \Semae\ScriptcaseOracle\FalhaOracleException: false/excecao do driver
// ou resposta de consulta invalida. A mensagem e controlada, sem SQL/binds.
// $falha->getCodigoOracle(): ?string => 'ORA-00942', por exemplo, ou null.
// A excecao bruta do driver nao e encadeada na excecao publica.
// O evento deve tratar a falha e interromper o fluxo dependente da operacao.

// Transacoes e limites:
// A biblioteca NAO inicia, confirma ou desfaz transacoes, nao fecha a conexao,
// nao exibe mensagens e nao grava logs. Essas decisoes cabem ao hospedeiro.
// Indicada para listas pequenas. Sem leitura progressiva, CLOB/BLOB, parametros
// de saida, ORM ou atalhos buscarUma()/buscarValor().

// ============================================================================
// EXEMPLO EXECUTAVEL - LEITURA
// ============================================================================
// Este par foi confirmado no ambiente do projeto. Ajuste ao reutilizar em outro.
$charsetConexao = 'ISO-8859-1';
$charsetAplicacao = 'ISO-8859-1';

if (!method_exists($this, 'criarConexaoOracle')) {
    echo 'Habilite conexaoOracle.php nas bibliotecas internas e gere a aplicacao.';
    return;
}

try {
    $db = $this->criarConexaoOracle($this->Db, $charsetConexao, $charsetAplicacao);

    // Apostrofo e acentos sao valores do bind, sem concatenacao no SQL.
    $texto = "D'Ávila - ação, café e órgão";
    $linhas = $db->buscarTodos(
        'SELECT :texto AS "TEXTO", :referencia AS "REFERENCIA" FROM DUAL',
        ['texto' => $texto, 'referencia' => 123]
    );

    // Consulta sem parametros e sem resultados: o contrato retorna [].
    $semRegistros = $db->buscarTodos('SELECT 1 AS "ID" FROM DUAL WHERE 1 = 0');

    echo '<h2>Exemplo da biblioteca Oracle</h2>';
    echo '<p>Consulta com binds:</p><pre>'
        . htmlspecialchars(var_export($linhas, true), ENT_QUOTES, $charsetAplicacao)
        . '</pre>';
    echo '<p>Consulta sem registros:</p><pre>'
        . htmlspecialchars(var_export($semRegistros, true), ENT_QUOTES, $charsetAplicacao)
        . '</pre>';
    echo '<p>Somente leitura. Nenhum INSERT, UPDATE, DELETE ou commit executado.</p>';
} catch (\Semae\ScriptcaseOracle\FalhaOracleException $falha) {
    echo 'Nao foi possivel consultar o Oracle.';
    if ($falha->getCodigoOracle() !== null) {
        echo ' Codigo: ' . htmlspecialchars($falha->getCodigoOracle(), ENT_QUOTES, $charsetAplicacao);
    }
    return;
} catch (\InvalidArgumentException $falha) {
    echo 'Confira a conexao, os charsets e os nomes e tipos dos parametros.';
    return;
} catch (\UnexpectedValueException $falha) {
    echo 'Texto invalido ou nao representavel no charset de destino.';
    return;
} catch (\Throwable $falha) {
    echo 'Nao foi possivel carregar ou executar o exemplo da biblioteca.';
    return;
}
