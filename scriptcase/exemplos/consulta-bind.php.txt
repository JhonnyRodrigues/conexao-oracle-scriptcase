// Primeiro teste de leitura: habilitar a interna pela flag e gerar a aplicacao.
// SQL e valor usam apenas ASCII. Os charsets efetivos serao confirmados antes dos testes com acentos.
try {
    $db = $this->criarConexaoOracle($this->Db, 'UTF-8', 'UTF-8');
    $valor = 'TESTE_BIND_123';

    $linhas = $db->buscarTodos(
        'SELECT :valor AS "VALOR" FROM DUAL',
        ['valor' => $valor]
    );

    $correto = count($linhas) === 1 && array_values($linhas[0]) === [$valor];
    echo $correto ? 'Consulta com bind: OK<br>' : 'Retorno diferente do esperado:<br>';
    echo '<pre>' . htmlspecialchars(var_export($linhas, true), ENT_QUOTES, 'UTF-8') . '</pre>';
} catch (\Semae\ScriptcaseOracle\FalhaOracleException $falha) {
    echo 'Falha na consulta: ' . htmlspecialchars($falha->getMessage(), ENT_QUOTES, 'UTF-8') . '<br>';
    echo 'Codigo Oracle: ' . ($falha->getCodigoOracle() ?? 'indisponivel');
    return;
}
