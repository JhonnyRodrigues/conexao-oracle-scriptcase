// Habilitar conexaoOracle.php em Bibliotecas Internas / Projeto e gerar a aplicacao.
// Apenas homologacao: mostra caminhos de arquivos e tenta construir o objeto, sem SQL.
echo 'PHP: ' . PHP_VERSION . '<br>';
echo 'mbstring: ' . (extension_loaded('mbstring') ? 'OK' : 'AUSENTE') . '<br>';
echo 'OCI8: ' . (extension_loaded('oci8') ? 'OK' : 'AUSENTE') . '<br>';

if (!method_exists($this, 'criarConexaoOracle')) {
    exit('Metodo criarConexaoOracle: AUSENTE. Confira a flag da biblioteca interna e gere novamente.');
}
echo 'Metodo criarConexaoOracle: OK<br>';

$classeOracle = 'Semae\\ScriptcaseOracle\\BancoOracle';
$raizAplicacoes = dirname(dirname($_SERVER['SCRIPT_FILENAME']));
$pastaExterna = $raizAplicacoes . '/_lib/libraries/sys/semae/conexaoOracleScriptcase';
echo 'Pasta esperada da externa: ' . htmlspecialchars($pastaExterna, ENT_QUOTES, 'UTF-8') . '<br>';
foreach (['autoload.php', 'vendor/autoload.php', 'src/BancoOracle.php', 'src/ConversorCharset.php', 'src/FalhaOracleException.php'] as $arquivo) {
    $caminho = $pastaExterna . '/' . $arquivo;
    echo htmlspecialchars($arquivo, ENT_QUOTES, 'UTF-8') . ': '
        . (is_file($caminho) ? 'EXISTE' : 'AUSENTE')
        . ' / ' . (is_readable($caminho) ? 'LEGIVEL' : 'NAO LEGIVEL') . '<br>';
}

$falhaCarregamento = null;
try {
    // Estes charsets servem apenas para construir o objeto neste teste sem SQL.
    // Antes de executar consultas, configure os charsets efetivos do ambiente.
    $db = $this->criarConexaoOracle($this->Db, 'UTF-8', 'UTF-8');
    echo 'Classe BancoOracle: ' . ($db instanceof \Semae\ScriptcaseOracle\BancoOracle ? 'OK' : 'AUSENTE') . '<br>';
    echo 'Construcao sem SQL: OK<br>';
} catch (\Throwable $falha) {
    $falhaCarregamento = $falha;
    // Diagnostico de instalacao em homologacao, antes de qualquer consulta ao driver.
    echo 'Falha de carregamento/configuracao: '
        . htmlspecialchars($falha->getMessage(), ENT_QUOTES, 'UTF-8') . '<br>';
}

$incluiuExterna = false;
$incluiuComposer = false;
foreach (get_included_files() as $arquivoIncluido) {
    $normalizado = str_replace('\\', '/', $arquivoIncluido);
    if (str_ends_with($normalizado, '/conexaoOracleScriptcase/autoload.php')) {
        $incluiuExterna = true;
    }
    if (str_ends_with($normalizado, '/conexaoOracleScriptcase/vendor/autoload.php')) {
        $incluiuComposer = true;
    }
}
echo 'Autoload externo incluido: ' . ($incluiuExterna ? 'SIM' : 'NAO') . '<br>';
echo 'Autoload Composer incluido: ' . ($incluiuComposer ? 'SIM' : 'NAO') . '<br>';

$encontrouPrefixo = false;
foreach (spl_autoload_functions() ?: [] as $carregador) {
    if (!is_array($carregador) || !is_object($carregador[0])) {
        continue;
    }
    $loader = $carregador[0];
    if (get_class($loader) !== 'Composer\\Autoload\\ClassLoader') {
        continue;
    }
    $pastas = $loader->getPrefixesPsr4()['Semae\\ScriptcaseOracle\\'] ?? [];
    if ($pastas === []) {
        continue;
    }
    $encontrouPrefixo = true;
    echo 'Namespace Oracle aponta para: '
        . htmlspecialchars(implode(', ', $pastas), ENT_QUOTES, 'UTF-8') . '<br>';
    $resolvido = $loader->findFile($classeOracle);
    echo 'Arquivo resolvido para BancoOracle: '
        . ($resolvido === false ? 'NAO ENCONTRADO' : htmlspecialchars($resolvido, ENT_QUOTES, 'UTF-8')) . '<br>';
}
echo 'Carregador do namespace Oracle registrado: ' . ($encontrouPrefixo ? 'SIM' : 'NAO') . '<br>';
if ($falhaCarregamento !== null) {
    return;
}
